import tkinter as tk from tkinter import ttk, messagebox from datetime import datetime # ============================================================ # CLASE GESTIONCLIENTES # ============================================================ class GestionClientes: def __init__( self, identificacion, nombre_completo, genero, tipo_menu, numero_sesiones, fecha_registro, costo_sesion ): self.identificacion = identificacion self.nombre_completo = nombre_completo self.genero = genero self.tipo_menu = tipo_menu self.numero_sesiones = numero_sesiones self.fecha_registro = fecha_registro self.costo_sesion = costo_sesion def calcular_costo_total(self): return self.numero_sesiones * self.costo_sesion # ============================================================ # VALORES DE LOS MENÚS # ============================================================ PRECIOS_MENU = { "Menú ejecutivo": 35000, "Menú vegetariano": 28000, "Menú degustación": 75000, "Menú infantil": 20000, "Menú gourmet": 95000 } # ============================================================ # FUNCIONES # ============================================================ def validar_acceso(): """ Valida la contraseña de acceso. La contraseña solicitada por el Anexo 1 es 1793. """ password = entrada_password.get() if password == "1793": ventana_login.destroy() mostrar_formulario() else: messagebox.showerror( "Acceso denegado", "La contraseña ingresada es incorrecta." ) def actualizar_precio(event=None): """ Actualiza automáticamente el costo de la sesión según el menú seleccionado. """ menu = combo_menu.get() if menu in PRECIOS_MENU: precio = PRECIOS_MENU[menu] entrada_costo.config(state="normal") entrada_costo.delete(0, tk.END) entrada_costo.insert(0, formatear_moneda(precio)) entrada_costo.config(state="readonly") def formatear_moneda(valor): """ Convierte un número a formato de pesos colombianos. """ return f"${valor:,.0f}".replace(",", ".") def guardar_registro(): """ Guarda la información del cliente en un objeto de la clase GestionClientes. """ try: identificacion = entrada_identificacion.get().strip() nombre = entrada_nombre.get().strip() genero = combo_genero.get() menu = combo_menu.get() sesiones = entrada_sesiones.get().strip() if not identificacion: messagebox.showwarning( "Dato requerido", "Ingrese la identificación." ) return if not nombre: messagebox.showwarning( "Dato requerido", "Ingrese el nombre completo." ) return if not genero: messagebox.showwarning( "Dato requerido", "Seleccione el género." ) return if not menu: messagebox.showwarning( "Dato requerido", "Seleccione el tipo de menú." ) return if not sesiones: messagebox.showwarning( "Dato requerido", "Ingrese el número de sesiones." ) return sesiones = int(sesiones) if sesiones <= 0: messagebox.showwarning( "Dato inválido", "El número de sesiones debe ser mayor que cero." ) return costo = PRECIOS_MENU[menu] fecha = datetime.now().strftime("%d/%m/%Y %H:%M:%S") global cliente_actual cliente_actual = GestionClientes( identificacion, nombre, genero, menu, sesiones, fecha, costo ) messagebox.showinfo( "Registro guardado", "Los datos del cliente fueron guardados correctamente." ) except ValueError: messagebox.showerror( "Error", "El número de sesiones debe ser un número entero." ) def mostrar_reporte(): """ Calcula el costo total y muestra el reporte. """ global cliente_actual if cliente_actual is None: messagebox.showwarning( "Sin registro", "Primero debe guardar el registro del cliente." ) return total = cliente_actual.calcular_costo_total() ventana_reporte = tk.Toplevel(ventana_principal) ventana_reporte.title("Reporte - Sabor & Sazón") ventana_reporte.geometry("600x520") ventana_reporte.resizable(False, False) titulo = tk.Label( ventana_reporte, text="SABOR & SAZÓN", font=("Arial", 20, "bold") ) titulo.pack(pady=15) subtitulo = tk.Label( ventana_reporte, text="Reporte del servicio gastronómico", font=("Arial", 12) ) subtitulo.pack(pady=5) marco = tk.Frame(ventana_reporte) marco.pack(padx=30, pady=20, fill="both") datos = [ ("Identificación:", cliente_actual.identificacion), ("Nombre completo:", cliente_actual.nombre_completo), ("Género:", cliente_actual.genero), ("Tipo de menú:", cliente_actual.tipo_menu), ("Número de sesiones:", str(cliente_actual.numero_sesiones)), ("Fecha de registro:", cliente_actual.fecha_registro), ("Costo por sesión:", formatear_moneda( cliente_actual.costo_sesion )), ("COSTO TOTAL:", formatear_moneda(total)) ] for i, (etiqueta, valor) in enumerate(datos): tk.Label( marco, text=etiqueta, font=("Arial", 11, "bold"), anchor="w" ).grid( row=i, column=0, sticky="w", padx=10, pady=8 ) tk.Label( marco, text=valor, font=("Arial", 11), anchor="w" ).grid( row=i, column=1, sticky="w", padx=10, pady=8 ) tk.Button( ventana_reporte, text="Cerrar reporte", command=ventana_reporte.destroy, width=20 ).pack(pady=15) def limpiar_formulario(): """ Limpia los datos ingresados. """ entrada_identificacion.delete(0, tk.END) entrada_nombre.delete(0, tk.END) combo_genero.set("") combo_menu.set("") entrada_sesiones.delete(0, tk.END) entrada_costo.config(state="normal") entrada_costo.delete(0, tk.END) entrada_costo.config(state="readonly") def salir_aplicacion(): """ Confirma si el usuario desea cerrar la aplicación. """ respuesta = messagebox.askyesno( "Salir", "¿Está seguro de que desea salir de la aplicación?" ) if respuesta: ventana_principal.destroy() # ============================================================ # FORMULARIO PRINCIPAL # ============================================================ def mostrar_formulario(): global ventana_principal global entrada_identificacion global entrada_nombre global combo_genero global combo_menu global entrada_sesiones global entrada_costo ventana_principal = tk.Tk() ventana_principal.title("Sabor & Sazón - Gestión de Clientes") ventana_principal.geometry("700x650") ventana_principal.resizable(False, False) #
